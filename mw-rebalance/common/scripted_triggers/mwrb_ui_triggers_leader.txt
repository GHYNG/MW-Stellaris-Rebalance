mwrb_is_leader_ruler = {
	is_scope_type = leader
	is_ruler = yes
}

mwrb_is_leader_governor_sector = {
	is_scope_type = leader
	exists = planet
	planet = {
		exists = leader
		is_sector_capital = yes
	}
}

mwrb_is_leader_governor_planet = {
	is_scope_type = leader
	exists = planet
	planet = {
		exists = leader
		is_sector_capital = no
	}
}

# fleet and army transport
mwrb_is_leader_fleet = {
	is_scope_type = leader
	exists = fleet
	leader_class = commander
	fleet = {
		exists = leader
		mwrb_equals = {
			A = leader
			B = Root
		}
	}
}

mwrb_is_leader_planet_army = {
	is_scope_type = leader
	exists = owner
	leader_class = commander
	NOT = {
		exists = fleet # as this army is on planet ground
	}
	owner = {
		any_owned_army = {
			exists = leader
			mwrb_equals = {
				A = leader
				B = Root
			}
		}
	}
}
